local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer
local Mouse = LocalPlayer:GetMouse()
local RunService = game:GetService("RunService")

local triggerbotEnabled = false
local lastShotTime = 0
local SHOT_COOLDOWN = 0.05

-- Función de clic con múltiples métodos
local function clickMouse()
    -- Método 1: mouse1click (más común)
    if mouse1click then
        mouse1click()
        return
    end
    
    -- Método 2: Synapse X
    if syn and syn.mouse1click then
        syn.mouse1click()
        return
    end
    
    -- Método 3: VirtualInputManager
    if VirtualInputManager then
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
        task.wait(0.05)
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
        return
    end
    
    -- Método 4: mouse_event (algunos ejecutores)
    if mouse_event then
        mouse_event(0x02, 0, 0, 0, 0)
        task.wait(0.05)
        mouse_event(0x04, 0, 0, 0, 0)
        return
    end
    
    -- Método 5: Simular con InputService (alternativa)
    local inputService = game:GetService("UserInputService")
    if inputService then
        inputService:SendMouseButtonEvent(Enum.UserInputType.MouseButton1, 0, 0, true)
        task.wait(0.05)
        inputService:SendMouseButtonEvent(Enum.UserInputType.MouseButton1, 0, 0, false)
        return
    end
end

-- Función para asegurar que el mouse está en la posición correcta
local function ensureMousePosition()
    local target = Mouse.Target
    if target then
        local position = target.Position
        if position then
            Mouse.Move(position)
        end
    end
end

UserInputService.InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.P then
        triggerbotEnabled = not triggerbotEnabled
        print("Triggerbot " .. (triggerbotEnabled and "ACTIVADO" or "DESACTIVADO"))
        if triggerbotEnabled then
            print("¡Apunta a un enemigo para disparar!")
        end
        lastShotTime = 0
    end
end)

RunService.RenderStepped:Connect(function()
    if not triggerbotEnabled then return end
    if not Mouse.Target then return end

    local target = Mouse.Target
    local character = target:FindFirstAncestorOfClass("Model")
    
    if character and character ~= LocalPlayer.Character then
        local humanoid = character:FindFirstChild("Humanoid")
        if humanoid and humanoid.Health > 0 then
            local now = tick()
            if now - lastShotTime >= SHOT_COOLDOWN then
                -- Asegurar que el mouse está sobre el objetivo
                ensureMousePosition()
                -- Disparar
                clickMouse()
                lastShotTime = now
            end
        end
    end
end)

print("Triggerbot cargado. Presiona P para activar/desactivar.")
print("Si no funciona, prueba con otro ejecutor o verifica los permisos.")
