local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer
local Mouse = LocalPlayer:GetMouse()
local RunService = game:GetService("RunService")

local triggerbotEnabled = false
local lastShotTime = 0
local SHOT_COOLDOWN = 0.01

-- Solo función de clic
local function clickMouse()
    if mouse1click then
        mouse1click()
    elseif syn and syn.mouse1click then
        syn.mouse1click()
    elseif VirtualInputManager then
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
        task.wait(0.01)
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
    end
end

UserInputService.InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.P then
        triggerbotEnabled = not triggerbotEnabled
        print("Triggerbot: " .. tostring(triggerbotEnabled))
        lastShotTime = 0
    end
end)

RunService.RenderStepped:Connect(function()
    if not triggerbotEnabled then return end
    if not Mouse.Target then return end

    local target = Mouse.Target
    local character = target:FindFirstAncestorOfClass("Model")
    
    if character and character ~= LocalPlayer.Character then
        local humanoid = character:FindFirstChild("Humanoid")
        if humanoid and humanoid.Health > 0 then
            local now = tick()
            if now - lastShotTime >= SHOT_COOLDOWN then
                clickMouse() -- Solo clic, sin presionar 2
                lastShotTime = now
            end
        end
    end
end)

print("Triggerbot cargado. Presiona P para activar.")
