	dotsLoopRunning = true

		while dotsLoopRunning do
			for i = 1, 3 do
				if not dotsLoopRunning then break end
				TweenService:Create(dots[i], TweenInfo.new(0.25), {Size = UDim2.fromOffset(11, 11)}):Play()
				task.wait(0.12)
				TweenService:Create(dots[i], TweenInfo.new(0.25), {Size = UDim2.fromOffset(8, 8)}):Play()

local function formatDuration(seconds)
	seconds = math.max(0, math.floor(seconds))
	local h = math.floor(seconds / 3600)
	local m = math.floor((seconds % 3600) / 60)
	local s = seconds % 60
	return string.format("%dh %dm %ds", h, m, s)
end

local function hideStatus()
	stopStatusLoop()
	statusLabel.Text = ""
	statusLabel.TextTransparency = 1
end

local function showKeyStatus(keyData)
	stopStatusLoop()
	statusLabel.TextTransparency = 1

	if keyData.type == "permanent" then
		statusLabel.Text = "PERMANENT  •  EXPIRES: NEVER"
	else
		statusLoopRunning = true
		task.spawn(function()
			while statusLoopRunning do
				local remaining = keyData.expiresAt - os.time()
				if remaining <= 0 then
					statusLabel.Text = "TEMPORARY  •  EXPIRED"
					statusLoopRunning = false
					break
				end
				statusLabel.Text = "TEMPORARY  •  " .. formatDuration(remaining)
				task.wait(1)
			end
		end)
	end

	TweenService:Create(statusLabel, TweenInfo.new(0.25), {TextTransparency = 0}):Play()
end

local verifying = false

local function verifyKey()
	if verifying then return end
	local inputText = keyBox.Text:gsub("%s+", ""):lower()

	hideStatus()

	if inputText == "" then
		showFeedback("escribí una key primero", COLOR_ERROR)
		shakeInput()
		return
	end

	verifying = true
	setLoading(true)
	feedback.Text = ""

	task.wait(0.9)

	local keyData = KEYS[inputText]

	if keyData then
		setLoading(false)
		showFeedback("¡key correcta! bienvenido", COLOR_SUCCESS)
		showKeyStatus(keyData)
		TweenService:Create(inputStroke, TweenInfo.new(0.2), {Color = COLOR_SUCCESS}):Play()
		task.wait(1.4)
		closeGui(true)
	else
		verifying = false
		setLoading(false)
		showFeedback("key incorrecta, intentá de nuevo", COLOR_ERROR)
		TweenService:Create(inputStroke, TweenInfo.new(0.2), {Color = COLOR_ERROR}):Play()
		shakeInput()
		task.wait(1.2)
		TweenService:Create(inputStroke, TweenInfo.new(0.3), {Color = Color3.fromRGB(65, 65, 75)}):Play()
	end
end
